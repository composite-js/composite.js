const sports = [
  "Athletics",
  "Swimming",
  "Gymnastics",
  "Cycling",
  "Rowing",
  "Fencing",
  "Boxing",
  "Wrestling",
  "Sailing",
  "Basketball",
];

const sportColors = [
  "#27a6c9",
  "#82c66f",
  "#f1c75b",
  "#ed6a5a",
  "#a78bdc",
  "#4cc9a6",
  "#f08a9d",
  "#b7d55a",
  "#66b7e8",
  "#ef9f5a",
];

const participation = years.flatMap((year, index) => {
  const total = Math.round(
    260 +
      index * 310 +
      Math.sin(index * 0.82) * 320 +
      (index > 18 ? (index - 18) * 120 : 0),
  );
  const femaleShare = Math.min(0.46, 0.025 + index * 0.015);
  const female = Math.round(total * femaleShare);
  const male = total - female;

  return [
    { year, athletes: male, gender: "Male" },
    { year, athletes: female, gender: "Female" },
  ];
});

const pacData = years.map((year, index) => ({
  year,
  count: Math.round(12 + index * 6 + Math.sin(index * 0.65) * 12),
}));

const flowData = [];
years.forEach((year, yearIndex) => {
  sports.forEach((sport, sportIndex) => {
    const introduced = Math.max(0, sportIndex - 5);
    if (yearIndex < introduced) {
      return;
    }

    const rhythm =
      Math.sin(yearIndex * 0.64 + sportIndex * 0.91) +
      Math.cos(yearIndex * 0.31 + sportIndex * 0.37);
    const count = Math.max(
      1,
      Math.round(2 + sportIndex * 0.35 + yearIndex * 0.16 + rhythm * 2.2),
    );

    flowData.push({ year, sport, count });
  });
});

const rowHeight = 22;
const bodyHeight = years.length * rowHeight;

const dumbbell = chart({
  mark: "dumbbell",
  data: participation,
  width: 330,
  height: bodyHeight,
  margin: { top: 38, right: 18, bottom: 28, left: 54 },
  encoding: { x: "athletes", y: "year", xDomain: [0, 9000], yDomain: years },
  colors: ["#1aa7c9", "#ff5d67"],
  lineColor: "rgba(230, 230, 230, 0.48)",
  lineWidth: 1.6,
  radius: 4,
  xAxisName: "Number of participants and gender gap",
  showYAxis: true,
  xAxisPos: "top",
  padding: { yInner: 0.22, yOuter: 0.05 },
});

const pac = chart({
  mark: "pac",
  data: pacData,
  width: 50,
  height: bodyHeight,
  margin: { top: 38, right: 10, bottom: 28, left: 12 },
  encoding: { x: "count", y: "year", yDomain: years },
  color: "#8ecae6",
  opacity: 0.54,
  showYAxis: false,
  showXAxis: false,
  padding: { yInner: 0.22, yOuter: 0.05 },
});

const flow = chart({
  mark: "flow",
  data: flowData,
  width: 430,
  height: bodyHeight,
  margin: { top: 38, right: 100, bottom: 28, left: 22 },
  encoding: {
    x: "year",
    group: "sport",
    y: "count",
    xDomain: years,
    groupDomain: sports,
  },
  colors: sportColors,
  colorBy: "group",
  opacity: 0.18,
  minStrokeWidth: 0.25,
  maxStrokeWidth: 4.8,
  xLabelName: "HOST CITY",
  groupLabelName: "SPORT",
  showGroupLabels: true,
  labelFill: "rgba(255, 255, 255, 0.78)",
  labelFontSize: 10,
});

const composite = stackX([dumbbell, pac, flow]);

return composite;
