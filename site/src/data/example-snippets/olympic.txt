const datasetUrls = {
  games: exampleAssetUrl("dataset/olympic/games.csv"),
  sports: exampleAssetUrl("dataset/olympic/sports.csv"),
};

const sportColors = [
  "#27a6c9",
  "#82c66f",
  "#f1c75b",
  "#ed6a5a",
  "#a78bdc",
  "#4cc9a6",
  "#f08a9d",
  "#b7d55a",
  "#66b7e8",
  "#ef9f5a",
];

function unique(rows, key) {
  return [...new Set(rows.map((row) => row[key]))];
}

function buildView({ games, sports: sportRows }) {
  const years = games.map((game) => String(game.year));
  const flowData = sportRows.map((row) => ({
    ...row,
    year: String(row.year),
  }));
  const sports = unique(flowData, "sport");
  const participation = games.flatMap((game) => [
    { year: String(game.year), athletes: game.male, gender: "Male" },
    { year: String(game.year), athletes: game.female, gender: "Female" },
  ]);
  const pacData = games.map((game) => ({
    year: String(game.year),
    count: game.pac,
  }));
  const rowHeight = 22;
  const bodyHeight = years.length * rowHeight;

  const dumbbell = chart({
    mark: "dumbbell",
    data: participation,
    width: 330,
    height: bodyHeight,
    margin: { top: 38, right: 18, bottom: 28, left: 54 },
    encoding: { x: "athletes", y: "year", xDomain: [0, 9000], yDomain: years },
    colors: ["#1aa7c9", "#ff5d67"],
    lineColor: "rgba(230, 230, 230, 0.48)",
    lineWidth: 1.6,
    radius: 4,
    xAxisName: "Number of participants and gender gap",
    showYAxis: true,
    xAxisPos: "top",
    padding: { yInner: 0.22, yOuter: 0.05 },
  });

  const pac = chart({
    mark: "pac",
    data: pacData,
    width: 50,
    height: bodyHeight,
    margin: { top: 38, right: 10, bottom: 28, left: 12 },
    encoding: { x: "count", y: "year", yDomain: years },
    color: "#8ecae6",
    opacity: 0.54,
    showYAxis: false,
    showXAxis: false,
    padding: { yInner: 0.22, yOuter: 0.05 },
  });

  const flow = chart({
    mark: "flow",
    data: flowData,
    width: 430,
    height: bodyHeight,
    margin: { top: 38, right: 100, bottom: 28, left: 22 },
    encoding: {
      x: "year",
      group: "sport",
      y: "count",
      xDomain: years,
      groupDomain: sports,
    },
    colors: sportColors,
    colorBy: "group",
    opacity: 0.18,
    minStrokeWidth: 0.25,
    maxStrokeWidth: 4.8,
    xLabelName: "HOST CITY",
    groupLabelName: "SPORT",
    showGroupLabels: true,
    labelFill: "rgba(255, 255, 255, 0.78)",
    labelFontSize: 10,
  });

  return stackX([dumbbell, pac, flow]);
}

const [games, sports] = await Promise.all(
  Object.values(datasetUrls).map(async (url) =>
    parseCsv(await loadCsvText(url)),
  ),
);

return buildView({ games, sports });
