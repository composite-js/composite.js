// System: DropoutSeer
// Title: DropoutSeer: Visualizing learning patterns in Massive Open Online Courses for dropout reasoning and prediction
// Authors: Yuanzhe Chen, Qing Chen, Mingqian Zhao, Sebastien Boyer, Kalyan Veeramachaneni, and Huamin Qu
// Link: https://ieeexplore.ieee.org/document/7883517
const datasetUrls = {
  learners: exampleAssetUrl("dataset/dropoutseer/learners.csv"),
  sequence: exampleAssetUrl("dataset/dropoutseer/sequence.csv"),
  flow: exampleAssetUrl("dataset/dropoutseer/flow.csv"),
  bars: exampleAssetUrl("dataset/dropoutseer/bars.csv"),
};

const activityTypes = ["video", "quiz", "forum", "assignment"];
const colors = ["#89d4c9", "#f7ea57", "#f47f6b", "#2f77b4"];

function unique(rows, key) {
  return [...new Set(rows.map((row) => row[key]))];
}

function buildView({ learners, sequence: sequenceRows, flowData, barData }) {
  const weeks = unique(sequenceRows, "week");
  const dates = unique(flowData, "date");
  const sequenceRecords = sequenceRows.map((row) => ({
    ...row,
    mix: activityTypes.map((category) => ({
      category,
      value: row[category],
    })),
  }));
  const missing = learners.flatMap((learner) =>
    weeks
      .filter((week) => week > learner.dropoutWeek)
      .map((week) => ({ learner: learner.id, week })),
  );
  const events = learners
    .filter((learner) => learner.dropoutWeek < weeks.at(-1))
    .map((learner) => ({
      learner: learner.id,
      week: learner.dropoutWeek,
      type: "dropout",
    }));
  const learnerDomain = learners.map((learner) => learner.id);

  const sequence = embed(
    sequenceContainer({
      width: 700,
      height: 640,
      xDomain: weeks,
      yDomain: learnerDomain,
      tracks: learnerDomain,
      missing,
      events,
      trackColor: "#a9a9a9",
      missingColor: "#8b8b8b",
    }),
    repeat(sequenceRecords, (record) =>
      chart({
        mark: "pie",
        width: 28,
        height: 28,
        margin: { top: 0, right: 0, bottom: 0, left: 0 },
        data: record.mix,
        encoding: { x: "category", y: "value" },
        colorScheme: colors,
        innerRadius: 8,
      }),
    ),
    { x: "week", y: "learner", key: "id", width: 28, height: 28 },
  );

  const flow = chart({
    mark: "flow",
    width: 280,
    height: 640,
    margin: { top: 0, right: 0, bottom: 0, left: 0 },
    data: flowData,
    encoding: {
      x: "learner",
      group: "date",
      y: "count",
      xDomain: learnerDomain,
      groupDomain: dates,
    },
    color: "#f2ad73",
    opacity: 0.12,
    minStrokeWidth: 0.35,
    maxStrokeWidth: 5,
  });

  const bars = chart({
    mark: "stackbar",
    width: 230,
    height: 640,
    margin: { top: 26, right: 20, bottom: 20, left: 58 },
    data: barData,
    encoding: { x: "count", y: "date", group: "segment" },
    colorScheme: ["#2d6fa8", "#2fb5c4", "#cdebdc"],
    xAxisPos: "top",
    xAxisName: "",
    showLabels: false,
  });

  return stackX([sequence, flow, bars], { margin: 10 });
}

const [learners, sequence, flowData, barData] = await Promise.all(
  Object.values(datasetUrls).map(async (url) =>
    parseCsv(await loadCsvText(url)),
  ),
);

return buildView({ learners, sequence, flowData, barData });
