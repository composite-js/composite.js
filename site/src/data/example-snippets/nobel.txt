const datasetUrls = {
  fields: exampleAssetUrl("dataset/nobel/fields.csv"),
  ages: exampleAssetUrl("dataset/nobel/ages.csv"),
  education: exampleAssetUrl("dataset/nobel/education.csv"),
};

const educationDomain = ["Doctoral", "Master", "Bachelor", "Other"];
const rowWidth = 900;
const rowHeight = 120;

function buildView({ fields, ages, education }) {
  const fieldNames = fields.map((field) => field.name);
  const fieldsByName = new Map(
    fields.map((field) => [
      field.name,
      {
        ...field,
        ages: ages.filter((row) => row.field === field.name),
        education: education.filter((row) => row.field === field.name),
      },
    ]),
  );

  function nobelRow(fieldName) {
    const field = fieldsByName.get(fieldName);
    const ageArea = chart({
      data: field.ages,
      mark: "area",
      encoding: {
        x: "year",
        y: "age",
        xDomain: [1900, 2025],
        yDomain: [0, 100],
      },
      width: 560,
      height: 72,
      margin: { top: 14, right: 18, bottom: 28, left: 64 },
      color: field.color,
      fillOpacity: 0.22,
      strokeWidth: 2,
      xAxisName: "Award year",
      yAxisName: field.name,
    });

    const educationBars = chart({
      data: field.education,
      mark: "bar",
      encoding: {
        x: "count",
        y: "degree",
        xDomain: [0, 25],
        yDomain: educationDomain,
      },
      width: 210,
      height: 72,
      margin: { top: 14, right: 48, bottom: 28, left: 72 },
      color: field.color,
      showLabels: true,
      showXAxis: true,
      showYAxis: true,
      xAxisName: "Education count",
    });

    return stackX([ageArea, educationBars]);
  }

  return repeatY(fieldNames, nobelRow, {
    width: rowWidth,
    height: rowHeight * fieldNames.length,
    paddingInner: 0.1,
    paddingOuter: 0,
  });
}

const [fields, ages, education] = await Promise.all(
  Object.values(datasetUrls).map(async (url) =>
    parseCsv(await loadCsvText(url)),
  ),
);

return buildView({ fields, ages, education });
